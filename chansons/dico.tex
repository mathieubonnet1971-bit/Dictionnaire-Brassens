% Commands
\newcommand{\entry}[6]{%
    \par
    \leavevmode
    \markboth{#1}{#1}%
    \begin{minipage}{\linewidth}
        \textbf{#1}\ {#2}\ \textit{#3}\ $\bullet$\ {#4} {#5} \textit{\color{gray}#6}%
    \end{minipage}
    \par
    \smallskip
}

\newcommand{\alphabetletter}[1]{%
    \ifstrequal{\firstxmark}{#1}{%
        \colorbox{alphabetbackground}{%
            \textcolor{black}{\strut\bfseries #1}%
        }%
    }{%
        \ifstrequal{\lastxmark}{#1}{%
            \colorbox{alphabetbackground}{%
                \textcolor{black}{\strut\bfseries #1}%
            }%
        }{%
            {\color{alphabetgray}\strut #1}%
        }%
    }%
}
