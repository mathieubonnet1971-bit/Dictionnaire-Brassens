% Extracting entry first character
\newcommand{\FirstChar}[1]{%
    \StrLeft{#1}{1}[\firstletter]%
    \StrRight{#1}{1}[\lastletter]%
    \firstletter
}

% Circle
\newcommand{\cornerCircle}{
\begin{tikzpicture}
\filldraw[color=brickred, fill=red!5, thick, opacity=0.7](-1,0) circle (0.44);
\end{tikzpicture}
}

% Defining Left and Right marks
\def \borderTextRight{\FirstChar{\rightmark}}
\def \borderTextLeft{\FirstChar{\leftmark}}

% Lettre colors
\newcommand{\LetterColor}[1]{%
    \ifcsname letter#1exists\endcsname
        \color{grey_120}%
    \else
        \color{grey_180}%
    \fi
}
